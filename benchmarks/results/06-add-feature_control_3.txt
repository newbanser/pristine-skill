我来帮你添加按优先级排序的功能。首先，我们需要修改多个文件来实现这个功能。

## 1. 首先修改 `src/utils.js`，添加排序相关的工具函数：

```jsx
// src/utils.js

// 定义优先级对应的权重值
export const PRIORITY_WEIGHTS = {
  high: 3,
  medium: 2,
  low: 1
};

// 按优先级排序
export function sortByPriority(todos, order = 'desc') {
  return [...todos].sort((a, b) => {
    const weightA = PRIORITY_WEIGHTS[a.priority] || 0;
    const weightB = PRIORITY_WEIGHTS[b.priority] || 0;
    
    if (order === 'desc') {
      return weightB - weightA;  // 高优先级在前
    }
    return weightA - weightB;  // 低优先级在前
  });
}

// 按创建时间排序（保留原有功能）
export function sortByCreatedAt(todos) {
  return [...todos].sort((a, b) => {
    return new Date(a.createdAt) - new Date(b.createdAt);
  });
}
```

## 2. 修改 `src/App.jsx`，添加排序状态和切换控件：

```jsx
// src/App.jsx
import React, { useState } from 'react';
import TodoList from './TodoList';
import { sortByPriority, sortByCreatedAt } from './utils';

function App() {
  const [todos, setTodos] = useState([]);
  const [sortType, setSortType] = useState('createdAt'); // 'createdAt' 或 'priority'
  const [priorityOrder, setPriorityOrder] = useState('desc'); // 'asc' 或 'desc'

  const addTodo = (text, priority = 'medium') => {
    const newTodo = {
      id: Date.now(),
      text,
      priority,
      completed: false,
      createdAt: new Date().toISOString()
    };
    setTodos([...todos, newTodo]);
  };

  // 根据排序类型处理待办事项
  const getSortedTodos = () => {
    let sortedTodos = [...todos];
    
    if (sortType === 'priority') {
      sortedTodos = sortByPriority(sortedTodos, priorityOrder);
    } else {
      sortedTodos = sortByCreatedAt(sortedTodos);
    }
    
    return sortedTodos;
  };

  // 切换排序类型
  const handleSortChange = (type) => {
    setSortType(type);
  };

  // 切换优先级排序方向
  const togglePriorityOrder = () => {
    setPriorityOrder(prev => prev === 'desc' ? 'asc' : 'desc');
  };

  return (
    <div className="app">
      <h1>待办事项</h1>
      
      {/* 排序控制区 */}
      <div className="sort-controls">
        <button 
          onClick={() => handleSortChange('createdAt')}
          className={sortType === 'createdAt' ? 'active' : ''}
        >
          按创建时间
        </button>
        <button 
          onClick={() => handleSortChange('priority')}
          className={sortType === 'priority' ? 'active' : ''}
        >
          按优先级
        </button>
        {sortType === 'priority' && (
          <button onClick={togglePriorityOrder}>
            {priorityOrder === 'desc' ? '高→低' : '低→高'}
          </button>
        )}
      </div>

      {/* 添加待办的表单 */}
      <TodoForm onAdd={addTodo} />

      {/* 待办列表 */}
      <TodoList 
        todos={getSortedTodos()} 
        setTodos={setTodos}
      />
    </div>
  );
}

// 添加表单组件（假设还没有）
function TodoForm({ onAdd }) {
  const [text, setText] = useState('');
  const [priority, setPriority] = useState('medium');

  const handleSubmit = (e) => {
    e.preventDefault();
    if (text.trim()) {
      onAdd(text, priority);
      setText('');
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <input 
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="添加新的待办事项"
      />
      <select 
        value={priority}
        onChange={(e) => setPriority(e.target.value)}
      >
        <option value="high">高优先级</option>
        <option value="medium">中优先级</option>
        <option value="low">低优先级</option>
      </select>
      <button type="submit">添加</button>
    </form>
  );
}

export default App;
```

import React from 'react';
import TodoItem from './TodoItem';

function TodoList({ todos, setTodos }) {
  return (
    <div className="todo-list">
      {todos.length === 0 ? (
        <p>暂无待办事项</p>
      ) : (
        todos.map(todo => (
          <TodoItem 
            key={todo.id}
            todo={todo}
            todos={todos}
            setTodos={setTodos}
          />
        ))
      )}
    </div>
  );
}

export default TodoList;

## 4. 修改 `src/TodoItem.jsx`，添加优先级显示和编辑功能：

```jsx
// src/TodoItem.jsx
import React, { useState } from 'react';

function TodoItem({ todo, todos, setTodos }) {
  const [isEditing, setIsEditing] = useState(false);
  const [editText, setEditText] = useState(todo.text);

  const toggleComplete = () => {
    setTodos(todos.map(t => 
      t.id === todo.id ? {...t, completed: !t.completed} : t
    ));
  };

  const deleteTodo = () => {
    setTodos(todos.filter(t => t.id !== todo.id));
  };

  const updateTodo = () => {
    setTodos(todos.map(t => 
      t.id === todo.id ? {...t, text: editText} : t
    ));
    setIsEditing(false);
  };

  const updatePriority = (newPriority) => {
    setTodos(todos.map(t => 
      t.id === todo.id ? {...t, priority: newPriority} : t
    ));
  };

  const priorityColors = {
    high: 'red',
    medium: 'orange',
    low: 'green'
  };

  return (
    <div className={`todo-item ${todo.completed ? 'completed' : ''}`}>
      <input 
        type="checkbox"
        checked={todo.completed}
        onChange={toggleComplete}
      />
      
      {isEditing ? (
        <>
          <input 
            value={editText}
            onChange={(e) => setEditText(e.target.value)}
          />
          <button onClick={updateTodo}>保存</button>
          <button onClick={() => setIsEditing(false)}>取消</button>
        </>
      ) : (
        <>
          <span>{todo.text}</span>
          <button onClick={() => setIsEditing(true)}>编辑</button>
        </>
      )}

      {/* 优先级选择器 */}
      <select 
        value={todo.priority}
        onChange={(e) => updatePriority(e.target.value)}
        style={{color: priorityColors[todo.priority]}}
      >
        <option value="low">低</option>
        <option value="medium">中</option>
        <option value="high">高</option>
      </select>

      <button onClick={deleteTodo}>删除</button>
    </div>
  );
}

export default TodoItem;
```
