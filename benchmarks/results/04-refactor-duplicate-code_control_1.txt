```javascript
// 提取日期格式化的通用函数
function formatDateWithSeparator(date, separator) {
  const d = new Date(date);
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${year}${separator}${month}${separator}${day}`;
}

// 基于通用函数创建特定格式的函数
function formatDate(date) {
  return formatDateWithSeparator(date, '-');
}

function formatDate2(date) {
  return formatDateWithSeparator(date, '/');
}

// 中文日期格式可以单独处理，因为它格式不同
function displayDate(date) {
  const d = new Date(date);
  return `${d.getMonth() + 1}月${d.getDate()}日`;
}
```

如果你想更简洁一些，也可以直接使用通用函数，不保留单独的包装函数：

```javascript
// 更简洁的版本
function formatDateWithSeparator(date, separator = '-') {
  const d = new Date(date);
  const parts = [
    d.getFullYear(),
    String(d.getMonth() + 1).padStart(2, '0'),
    String(d.getDate()).padStart(2, '0')
  ];
  return parts.join(separator);
}

// 使用方式
const date1 = formatDateWithSeparator(new Date(), '-');  // YYYY-MM-DD
const date2 = formatDateWithSeparator(new Date(), '/');  // YYYY/MM/DD
```

**其他改进建议：**

如果你需要处理更多格式，可以考虑使用一个更通用的函数：

```javascript
// 通用的格式化函数
function formatDate(date, options = {}) {
  const d = new Date(date);
  const {
    separator = '-',
    yearFirst = true,
    includeWeekday = false
  } = options;
  
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  
  let result;
  if (yearFirst) {
    result = `${year}${separator}${month}${separator}${day}`;
  } else {
    result = `${day}${separator}${month}${separator}${year}`;
  }
  
  if (includeWeekday) {
    const weekdays = ['日', '一', '二', '三', '四', '五', '六'];
    result += ` 星期${weekdays[d.getDay()]}`;
  }
  
  return result;
}

// 使用示例
const date1 = formatDate(new Date(), { separator: '-' });                    // 2023-12-25
const date2 = formatDate(new Date(), { separator: '/', yearFirst: false }); // 25/12/2023
const date3 = formatDate(new Date(), { includeWeekday: true });             // 2023-12-25 星期一
```

这样重构后，代码更加灵活、可维护，并且消除了重复的日期处理逻辑。
