**问题**：`formatDate` 和 `formatDate2` 重复度最高。`displayDate` 也重复了解析逻辑。修复：抽出 `parseDate` 和接受分隔符的 `formatYMD`。

```typescript
function parseDate(date: string | Date): Date {
  return date instanceof Date ? date : new Date(date);
}

function formatYMD(date: string | Date, sep = '-'): string {
  const d = parseDate(date);
  return `${d.getFullYear()}${sep}${String(d.getMonth()+1).padStart(2,'0')}${sep}${String(d.getDate()).padStart(2,'0')}`;
}

function formatDate(date: string | Date): string {
  return formatYMD(date);
}

function formatDate2(date: string | Date): string {
  return formatYMD(date, '/');
}

function displayDate(date: string | Date): string {
  const d = parseDate(date);
  return `${d.getMonth()+1}月${d.getDate()}日`;
}
```
