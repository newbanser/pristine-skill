## 1. `src/utils.js` - 添加优先级排序工具函数

```javascript
// 优先级排序工具函数
export const PRIORITY_ORDER = {
  high: 0,
  medium: 1,
  low: 2
};

// 按优先级排序
export const sortByPriority = (todos) => {
  return [...todos].sort((a, b) => {
    // 先按优先级排序
    const priorityDiff = (PRIORITY_ORDER[a.priority] || 2) - (PRIORITY_ORDER[b.priority] || 2);
    if (priorityDiff !== 0) {
      return priorityDiff;
    }
    // 如果优先级相同，按创建时间排序（新的在前）
    return new Date(b.createdAt) - new Date(a.createdAt);
  });
};

// 按创建时间排序（保持原有功能）
export const sortByCreatedAt = (todos) => {
  return [...todos].sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
};
```

## 2. `src/App.jsx` - 添加排序状态和控制

```jsx
import React, { useState, useEffect } from 'react';
import TodoList from './TodoList';
import { sortByPriority, sortByCreatedAt } from './utils';

function App() {
  const [todos, setTodos] = useState([]);
  const [sortMode, setSortMode] = useState('createdAt'); // 'createdAt' 或 'priority'
  const [newTodo, setNewTodo] = useState('');

  // 根据排序模式获取排序后的todos
  const getSortedTodos = () => {
    if (sortMode === 'priority') {
      return sortByPriority(todos);
    }
    return sortByCreatedAt(todos);
  };

  const handleAddTodo = (e) => {
    e.preventDefault();
    if (!newTodo.trim()) return;
    
    const newTodoItem = {
      id: Date.now(),
      text: newTodo,
      completed: false,
      priority: 'medium',
      createdAt: new Date()
    };
    
    setTodos([...todos, newTodoItem]);
    setNewTodo('');
  };

  const handleToggleTodo = (id) => {
    setTodos(todos.map(todo => 
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ));
  };

  const handleDeleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };

  const handleUpdatePriority = (id, priority) => {
    setTodos(todos.map(todo =>
      todo.id === id ? { ...todo, priority } : todo
    ));
  };

  return (
    <div className="app">
      <h1>待办事项</h1>
      
      {/* 排序模式切换 */}
      <div className="sort-controls">
        <label>排序方式：</label>
        <select value={sortMode} onChange={(e) => setSortMode(e.target.value)}>
          <option value="createdAt">按创建时间</option>
          <option value="priority">按优先级</option>
        </select>
      </div>

      {/* 添加新待办 */}
      <form onSubmit={handleAddTodo} className="add-todo-form">
        <input
          type="text"
          value={newTodo}
          onChange={(e) => setNewTodo(e.target.value)}
          placeholder="输入新待办事项"
        />
        <button type="submit">添加</button>
      </form>

      {/* 待办列表 */}
      <TodoList
        todos={getSortedTodos()}
        onToggle={handleToggleTodo}
        onDelete={handleDeleteTodo}
        onUpdatePriority={handleUpdatePriority}
      />
    </div>
  );
}

export default App;
```

## 3. `src/TodoItem.jsx` - 添加优先级显示和修改功能

```jsx
import React from 'react';

// 优先级对应的样式和标签
const PRIORITY_CONFIG = {
  high: { label: '高', color: '#dc3545' },
  medium: { label: '中', color: '#ffc107' },
  low: { label: '低', color: '#28a745' }
};

function TodoItem({ todo, onToggle, onDelete, onUpdatePriority }) {
  return (
    <li className={`todo-item ${todo.completed ? 'completed' : ''}`}>
      <div className="todo-content">
        <input
          type="checkbox"
          checked={todo.completed}
          onChange={() => onToggle(todo.id)}
        />
        <span className="todo-text">{todo.text}</span>
      </div>
      
      <div className="todo-actions">
        {/* 优先级显示和修改 */}
        <select
          value={todo.priority || 'medium'}
          onChange={(e) => onUpdatePriority(todo.id, e.target.value)}
          className="priority-select"
          style={{ backgroundColor: PRIORITY_CONFIG[todo.priority]?.color || '#ffc107' }}
        >
          <option value="high">高优先级</option>
          <option value="medium">中优先级</option>
          <option value="low">低优先级</option>
        </select>
        
        <button onClick={() => onDelete(todo.id)} className="delete-btn">
          删除
        </button>
      </div>
    </li>
  );
}

export default TodoItem;
```

## 4. `src/TodoList.jsx` - 传递新的处理函数

```jsx
import React from 'react';
import TodoItem from './TodoItem';

function TodoList({ todos, onToggle, onDelete, onUpdatePriority }) {
  return (
    <ul className="todo-list">
      {todos.length > 0 ? (
        todos.map(todo => (
          <TodoItem
            key={todo.id}
            todo={todo}
            onToggle={onToggle}
            onDelete={onDelete}
            onUpdatePriority={onUpdatePriority}
          />
        ))
      ) : (
        <li className="empty-list">暂无待办事项</li>
      )}
    </ul>
  );
}

export default TodoList;
```

## 主要改动说明：

1. **新增数据支持**：每个待办事项现在有 `priority` 属性（high/medium/low）和 `createdAt` 时间戳
2. **排序功能**：添加了 `sortByPriority` 和 `sortByCreatedAt` 两个工具函数
3. **界面控制**：添加排序方式选择器，默认按创建时间排序
4. **优先级修改**：可以在每个待办事项上直接修改优先级
5. **视觉反馈**：优先级通过不同颜色的下拉菜单显示（红=高，黄=中，绿=低）

这样用户就可以：
- 选择按优先级排序（高→中→低）
- 随时修改每个待办事项的优先级
- 切换回按创建时间排序
