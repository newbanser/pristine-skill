我来帮您实现按优先级排序的功能。首先，我们需要了解当前的代码结构，然后进行修改。

**1. 首先，修改工具函数 `src/utils.js`：**

```javascript
// 优先级配置
export const PRIORITY_LEVELS = {
  HIGH: { value: 3, label: '高', color: '#ff4d4f' },
  MEDIUM: { value: 2, label: '中', color: '#faad14' },
  LOW: { value: 1, label: '低', color: '#52c41a' }
};

export const PRIORITY_OPTIONS = [
  { value: 'HIGH', label: '高优先级' },
  { value: 'MEDIUM', label: '中优先级' },
  { value: 'LOW', label: '低优先级' }
];

// 优先级排序函数
export const sortByPriority = (todos) => {
  return [...todos].sort((a, b) => {
    const priorityA = a.priority ? PRIORITY_LEVELS[a.priority].value : 0;
    const priorityB = b.priority ? PRIORITY_LEVELS[b.priority].value : 0;
    
    // 如果优先级相同，按创建时间排序（新的在前）
    if (priorityA === priorityB) {
      return new Date(b.createdAt) - new Date(a.createdAt);
    }
    
    return priorityB - priorityA; // 高优先级在前
  });
};

// 保持原有的排序函数（如果存在）
export const sortByCreatedAt = (todos) => {
  return [...todos].sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
};
```

**2. 修改主组件 `src/App.jsx`：**

```jsx
import React, { useState } from 'react';
import TodoList from './TodoList';
import { PRIORITY_OPTIONS } from './utils';

function App() {
  const [todos, setTodos] = useState([]);
  const [sortType, setSortType] = useState('createdAt'); // 'createdAt' 或 'priority'
  const [newTodo, setNewTodo] = useState('');
  const [newTodoPriority, setNewTodoPriority] = useState('MEDIUM');

  // 添加待办事项
  const addTodo = (e) => {
    e.preventDefault();
    if (!newTodo.trim()) return;
    
    const todo = {
      id: Date.now(),
      text: newTodo,
      complete: false,
      priority: newTodoPriority,
      createdAt: new Date()
    };
    
    setTodos([...todos, todo]);
    setNewTodo('');
    setNewTodoPriority('MEDIUM');
  };

  // 切换完成状态
  const toggleTodo = (id) => {
    setTodos(todos.map(todo => 
      todo.id === id ? { ...todo, complete: !todo.complete } : todo
    ));
  };

  // 删除待办事项
  const deleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };

  return (
    <div className="app">
      <h1>待办事项</h1>
      
      {/* 排序控制 */}
      <div className="sort-controls">
        <label>排序方式：</label>
        <select value={sortType} onChange={(e) => setSortType(e.target.value)}>
          <option value="createdAt">按创建时间</option>
          <option value="priority">按优先级</option>
        </select>
      </div>
      
      {/* 添加表单 */}
      <form onSubmit={addTodo}>
        <input
          type="text"
          value={newTodo}
          onChange={(e) => setNewTodo(e.target.value)}
          placeholder="输入新待办事项"
        />
        <select value={newTodoPriority} onChange={(e) => setNewTodoPriority(e.target.value)}>
          {PRIORITY_OPTIONS.map(option => (
            <option key={option.value} value={option.value}>
              {option.label}
            </option>
          ))}
        </select>
        <button type="submit">添加</button>
      </form>
      
      {/* 待办列表 */}
      <TodoList 
        todos={todos}
        sortType={sortType}
        onToggle={toggleTodo}
        onDelete={deleteTodo}
      />
    </div>
  );
}

export default App;
```

**3. 修改列表组件 `src/TodoList.jsx`：**

```jsx
import React from 'react';
import TodoItem from './TodoItem';
import { sortByPriority, sortByCreatedAt } from './utils';

function TodoList({ todos, sortType, onToggle, onDelete }) {
  // 根据排序类型选择排序函数
  const getSortedTodos = () => {
    if (sortType === 'priority') {
      return sortByPriority(todos);
    }
    return sortByCreatedAt(todos);
  };

  const sortedTodos = getSortedTodos();

  if (sortedTodos.length === 0) {
    return <p className="empty-message">暂无待办事项</p>;
  }

  return (
    <ul className="todo-list">
      {sortedTodos.map(todo => (
        <TodoItem
          key={todo.id}
          todo={todo}
          onToggle={onToggle}
          onDelete={onDelete}
        />
      ))}
    </ul>
  );
}

export default TodoList;
```

**4. 修改单项组件 `src/TodoItem.jsx`：**

```jsx
import React from 'react';
import { PRIORITY_LEVELS } from './utils';

function TodoItem({ todo, onToggle, onDelete }) {
  const priority = todo.priority ? PRIORITY_LEVELS[todo.priority] : null;

  return (
    <li className={`todo-item ${todo.complete ? 'completed' : ''}`}>
      <div className="todo-content">
        <label className="checkbox-label">
          <input
            type="checkbox"
            checked={todo.complete}
            onChange={() => onToggle(todo.id)}
          />
          <span className="todo-text">{todo.text}</span>
        </label>
        
        {priority && (
          <span 
            className="priority-badge"
            style={{ backgroundColor: priority.color }}
          >
            {priority.label}
          </span>
        )}
      </div>
      
      <button 
        className="delete-btn"
        onClick={() => onDelete(todo.id)}
      >
        ×
      </button>
    </li>
  );
}

export default TodoItem;
```
