  const rules = [
    [!user.name, "name required"],
    [user.name && user.name.length < 3, "name too short"],

    [user.email && user.email.indexOf("@") < 0, "email invalid"]
  ];
  const violation = rules.find(([failed]) => failed);
  return violation ? violation[1] : "ok";
}
