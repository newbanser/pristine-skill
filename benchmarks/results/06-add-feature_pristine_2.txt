```javascript
import { TodoItem } from './TodoItem';

// 排序逻辑：按优先级从高到低（假设 priority 字段，数字越大优先级越高）
const sortedTodos = [...todoList].sort((a, b) => b.priority - a.priority);

// 渲染

  <div>
    {sortedTodos.map(todo => (

    ))}
  </div>
);
```
