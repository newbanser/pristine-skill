## 3. database.js - 数据库配置

```javascript
const sqlite3 = require('sqlite3').verbose();
const path = require('path');
const bcrypt = require('bcrypt');

// 创建数据库连接
const db = new sqlite3.Database(path.join(__dirname, 'user.db'), (err) => {
    if (err) {
        console.error('数据库连接失败:', err.message);
    } else {
        console.log('已连接到SQLite数据库');
        initDatabase();
    }
});

// 初始化数据库
function initDatabase() {
    db.serialize(() => {
        // 创建用户表
        db.run(`
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT NOT NULL UNIQUE,
                email TEXT NOT NULL UNIQUE,
                password_hash TEXT NOT NULL,
                created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                last_login DATETIME,
                is_active INTEGER DEFAULT 1
            )
        `, (err) => {
            if (err) {
                console.error('创建表失败:', err.message);
            } else {
                console.log('用户表已准备好');
                // 创建默认管理员账户（可选）
                createDefaultAdmin();
            }
        });
    });
}

// 创建默认管理员账户
async function createDefaultAdmin() {
    const username = 'admin';
    const email = 'admin@example.com';
    const password = 'admin123';
    
    db.get('SELECT id FROM users WHERE username = ?', [username], async (err, row) => {
        if (!row) {
            try {
                const salt = await bcrypt.genSalt(10);
                const hash = await bcrypt.hash(password, salt);
                
                db.run(
                    'INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)',
                    [username, email, hash],
                    (err) => {
                        if (err) {
                            console.error('创建默认管理员失败:', err.message);
                        } else {
                            console.log('默认管理员已创建 (admin/admin123)');
                        }
                    }
                );
            } catch (err) {
                console.error('生成密码哈希失败:', err.message);
            }
        }
    });
}

module.exports = db;
```

## 5. routes/auth.js - 认证路由

```javascript
const express = require('express');
const router = express.Router();
const bcrypt = require('bcrypt');
const db = require('../database');
const { isAuthenticated, isNotAuthenticated } = require('../middleware/auth');

// 显示登录页面
router.get('/login', isNotAuthenticated, (req, res) => {
    res.sendFile('login.html', { root: './public' });
});

// 处理登录请求
router.post('/login', (req, res) => {
    const { username, password } = req.body;
    
    if (!username || !password) {
        return res.status(400).json({ 
            success: false, 
            message: '请提供用户名和密码' 
        });
    }
    
    db.get('SELECT * FROM users WHERE username = ?', [username], async (err, user) => {
        if (err) {
            console.error('数据库查询错误:', err);
            return res.status(500).json({ 
                success: false, 
                message: '服务器错误' 
            });
        }
        
        if (!user) {
            return res.status(401).json({ 
                success: false, 
                message: '用户名或密码不正确' 
            });
        }
        
        if (!user.is_active) {
            return res.status(403).json({ 
                success: false, 
                message: '账户已被禁用' 
            });
        }
        
        try {
            // 验证密码
            const isMatch = await bcrypt.compare(password, user.password_hash);
            
            if (!isMatch) {
                return res.status(401).json({ 
                    success: false, 
                    message: '用户名或密码不正确' 
                });
            }
            
            // 更新最后登录时间
            db.run('UPDATE users SET last_login = CURRENT_TIMESTAMP WHERE id = ?', 
                [user.id], (updateErr) => {
                if (updateErr) console.error('更新登录时间失败:', updateErr);
            });
            
            // 设置会话
            req.session.userId = user.id;
            req.session.username = user.username;
            
            res.json({ 
                success: true, 
                message: '登录成功',
                user: {
                    id: user.id,
                    username: user.username,
                    email: user.email
                }
            });
            
        } catch (bcryptErr) {
            console.error('密码验证错误:', bcryptErr);
            res.status(500).json({ 
                success: false, 
                message: '服务器错误' 
            });
        }
    });
});

// 显示注册页面
router.get('/register', isNotAuthenticated, (req, res) => {
    res.sendFile('register.html', { root: './public' });
});

// 处理注册请求
router.post('/register', async (req, res) => {
    const { username, email, password, confirmPassword } = req.body;
    
    // 验证输入
    if (!username || !email || !password || !confirmPassword) {
        return res.status(400).json({ 
            success: false, 
            message: '请填写所有字段' 
        });
    }
    
    if (password.length < 6) {
        return res.status(400).json({ 
            success: false, 
            message: '密码长度至少6个字符' 
        });
    }
    
    if (password !== confirmPassword) {
        return res.status(400).json({ 
            success: false, 
            message: '两次输入的密码不一致' 
        });
    }
    
    // 检查用户名是否已存在
    db.get('SELECT id FROM users WHERE username = ?', [username], async (err, user) => {
        if (user) {
            return res.status(409).json({ 
                success: false, 
                message: '用户名已存在' 
            });
        }
        
        // 检查邮箱是否已存在
        db.get('SELECT id FROM users WHERE email = ?', [email], async (err, user) => {
            if (user) {
                return res.status(409).json({ 
                    success: false, 
                    message: '邮箱已被注册' 
                });
            }
            
            try {
                // 加密密码
                const salt = await bcrypt.genSalt(10);
                const hash = await bcrypt.hash(password, salt);
                
                // 插入新用户
                db.run(
                    'INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)',
                    [username, email, hash],
                    function(err) {
                        if (err) {
                            console.error('注册失败:', err);
                            return res.status(500).json({ 
                                success: false, 
                                message: '注册失败，请稍后重试' 
                            });
                        }
                        
                        res.status(201).json({ 
                            success: true, 
                            message: '注册成功！',
                            userId: this.lastID
                        });
                    }
                );
            } catch (hashErr) {
                console.error('密码加密失败:', hashErr);
                res.status(500).json({ 
                    success: false, 
                    message: '服务器错误' 
                });
            }
        });
    });
});

// 退出登录
router.post('/logout', (req, res) => {
    if (req.session) {
        req.session.destroy(err => {
            if (err) {
                return res.status(500).json({ 
                    success: false, 
                    message: '退出失败' 
                });
            }
            res.clearCookie('connect.sid');
            res.json({ 
                success: true, 
                message: '已成功退出登录' 
            });
        });
    } else {
        res.json({ 
            success: true, 
            message: '已成功退出登录' 
        });
    }
});

// 获取当前登录用户信息
router.get('/current-user', isAuthenticated, (req, res) => {
    db.get(
        'SELECT id, username, email, created_at, last_login FROM users WHERE id = ?',
        [req.session.userId],
        (err, user) => {
            if (err || !user) {
                return res.status(404).json({ 
                    success: false, 
                    message: '用户不存在' 
                });
            }
            res.json({ 
                success: true, 
                user 
            });
        }
    );
});

// 显示仪表盘
router.get('/dashboard', isAuthenticated, (req, res) => {
    res.sendFile('dashboard.html', { root: './public' });
});

module.exports = router;
```

## 6. server.js - 主服务器文件

```javascript
const express = require('express');
const session = require('express-session');
const cookieParser = require('cookie-parser');
const path = require('path');
const authRoutes = require('./routes/auth');
const db = require('./database');

const app = express();
const PORT = process.env.PORT || 3000;

// 中间件
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(cookieParser());

// 配置会话
app.use(session({
    secret: 'your-secret-key-change-this-in-production',
    resave: false,
    saveUninitialized: false,
    cookie: {
        secure: false, // 在生产环境设置为true（HTTPS）
        httpOnly: true,
        maxAge: 24 * 60 * 60 * 1000 // 24小时
    }
}));

// 静态文件
app.use(express.static('public'));

// 认证路由
app.use('/', authRoutes);

// 首页
app.get('/', (req, res) => {
    if (req.session && req.session.userId) {
        res.redirect('/dashboard');
    } else {
        res.redirect('/login');
    }
});

// API 健康检查
app.get('/api/health', (req, res) => {
    res.json({ 
        status: 'ok', 
        message: '服务器运行正常' 
    });
});

// 错误处理中间件
app.use((err, req, res, next) => {
    console.error('服务器错误:', err);
    res.status(500).json({ 
        success: false, 
        message: '服务器内部错误' 
    });
});

// 启动服务器
app.listen(PORT, () => {
    console.log(`服务器已启动: http://localhost:${PORT}`);
});

// 关闭数据库连接
process.on('SIGINT', () => {
    db.close((err) => {
        if (err) {
            console.error('关闭数据库失败:', err.message);
        }
        console.log('数据库连接已关闭');
        process.exit(0);
    });
});
```
