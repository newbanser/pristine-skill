这里有几个不同层次的重构方案，从简单到更灵活：

## 方案1：使用参数控制分隔符

```javascript
function formatDate(date, separator = '-') {
  const d = new Date(date);
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${d.getFullYear()}${separator}${month}${separator}${day}`;
}

// 使用
formatDate(date);              // '2024-01-01'
formatDate(date, '/');         // '2024/01/01'

function displayDate(date) {
  const d = new Date(date);
  return `${d.getMonth() + 1}月${d.getDate()}日`;
}
```

## 方案2：使用配置对象（最灵活）

```javascript
function formatDate(date, options = {}) {
  const d = new Date(date);
  const { separator = '-', yearFirst = true, showYear = true } = options;
  
  const month = `${d.getMonth() + 1}月`;
  const day = `${d.getDate()}日`;
  const year = `${d.getFullYear()}年`;
  
  if (!showYear) return `${month}${day}`;
  return yearFirst 
    ? `${year}${separator}${String(d.getMonth() + 1).padStart(2, '0')}${separator}${String(d.getDate()).padStart(2, '0')}`
    : `${month}${day}`;
}
```

## 方案3：使用Intl.DateTimeFormat（推荐）

```javascript
// 最简单的方案，利用浏览器内置功能
const formatDate = (date, separator = '-') => {
  const d = new Date(date);
  const parts = {
    year: d.getFullYear(),
    month: String(d.getMonth() + 1).padStart(2, '0'),
    day: String(d.getDate()).padStart(2, '0')
  };
  return `${parts.year}${separator}${parts.month}${separator}${parts.day}`;
};

const formatDisplayDate = (date) => {
  const d = new Date(date);
  return `${d.getMonth() + 1}月${d.getDate()}日`;
};
```

## 方案4：工具函数组合（最优雅）

```javascript
// 基础工具函数
const pad = (num) => String(num).padStart(2, '0');

// 统一的日期解析函数
const parseDate = (date) => {
  const d = new Date(date);
  return {
    year: d.getFullYear(),
    month: d.getMonth() + 1,
    day: d.getDate()
  };
};

// 格式化函数
const formatDate = (date, separator = '-') => {
  const { year, month, day } = parseDate(date);
  return `${year}${separator}${pad(month)}${separator}${pad(day)}`;
};

const displayDate = (date) => {
  const { month, day } = parseDate(date);
  return `${month}月${day}日`;
};

// 使用方式保持不变
```

## 推荐方案

**如果只是解决重复问题**，推荐**方案1**，简单直接：
```javascript
function formatDate(date, separator = '-') {
  const d = new Date(date);
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${d.getFullYear()}${separator}${month}${separator}${day}`;
}
```

**如果需要更全面的日期处理**，推荐**方案3**或**方案4**，特别是方案4的代码组织方式更清晰，且便于后续扩展更多日期格式。
