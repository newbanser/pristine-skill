# 用户登录功能

搭建 Express + SQLite（better-sqlite3）登录，含注册、登录、JWT 认证。

## 项目结构

```
pristine-DEMO/
├── package.json         # 依赖声明
├── src/
│   ├── server.js        # 入口：启动 HTTP 服务
│   ├── db.js            # 数据库连接与建表
│   ├── routes/
│   │   └── auth.js      # 登录取证 / 登录验证 / 注销
│   └── middleware/
│       └── auth.js      # JWT 校验保护路由
```

**src/db.js** — 连接 + 建表（上线初稿：直接建表不用迁移）

```js
import Database from 'better-sqlite3';

const db = new Database('users.db');

// 上线前无存量用户 → 直接建表，无迁移函数
db.exec(`
  CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
  );
`);

export default db;
```

**src/routes/auth.js** — 注册 / 登录 / 注销

```js
import { Router } from 'express';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import db from '../db.js';

const router = Router();
const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret-change-me';
const TOKEN_EXPIRY = '7d';

// 注册：创建用户
router.post('/register', (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) 
    return res.status(400).json({ error: 'email 和 password 必填' });

  const password_hash = bcrypt.hashSync(password, 10);
  
  try {
    const result = db.prepare(
      'INSERT INTO users (email, password_hash) VALUES (?, ?)'
    ).run(email, password_hash);
    res.status(201).json({ id: result.lastInsertRowid });
  } catch (err) {
    if (err.code === 'SQLITE_CONSTRAINT_UNIQUE')
      return res.status(409).json({ error: '邮箱已注册' });
    throw err;
  }
});

// 登录：验证 → 签发 JWT
router.post('/login', (req, res) => {
  const { email, password } = req.body;
  const user = db.prepare(
    'SELECT * FROM users WHERE email = ?'
  ).get(email);

  if (!user || !bcrypt.compareSync(password, user.password_hash))
    return res.status(401).json({ error: '邮箱或密码错误' });

  const token = jwt.sign(
    { userId: user.id, email: user.email },
    JWT_SECRET,
    { expiresIn: TOKEN_EXPIRY }
  );
  res.json({ token });
});

// 注销：客户端删除 token 即完成，这里仅作占位（JWT 无状态）
router.post('/logout', (req, res) => {
  res.json({ ok: true });
});

export default router;
```

**src/middleware/auth.js** — 保护路由

```js
import jwt from 'jsonwebtoken';

export function requireAuth(req, res, next) {
  const header = req.headers.authorization;
  const token = header?.startsWith('Bearer ') ? header.slice(7) : null;

  if (!token) return res.status(401).json({ error: '未提供 token' });

  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET || 'dev-secret-change-me');
    next();
  } catch {
    res.status(401).json({ error: 'token 无效或已过期' });
  }
}
```

**src/server.js**

```js
import express from 'express';
import authRouter from './routes/auth.js';
import { requireAuth } from './middleware/auth.js';

const app = express();
app.use(express.json());

app.use('/api/auth', authRouter);

// 受保护示例：仅在登录后可达
app.get('/api/me', requireAuth, (req, res) => {
  res.json({ user: req.user });
});

app.listen(3000, () => 
  console.log('API 运行在 http://localhost:3000')
);
```

## 测试（输出纯净：只读"读"第三行）

用 curl 试登录流程：

```bash
# 注册
curl -X POST http://localhost:3000/api/auth/register -H "Content-Type: application/json" -d '{"email":"a@test.com","password":"123456"}'
# 返回：{"id":1}

# 登录拿 token
curl -X POST http://localhost:3000/api/auth/login -H "Content-Type: application/json" -d '{"email":"a@test.com","password":"123456"}'
# 返回：{"token":"eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."}

# 用 token 访问 /api/me
curl http://localhost:3000/api/me -H "Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
# 返回：{"user":{"userId":1,"email":"a@test.com","iat":1712345678,"exp":1712950478}}
```
