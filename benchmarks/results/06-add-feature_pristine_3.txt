现在就在 TodoList.jsx 中修改排序逻辑，并在 utils.js 中添加排序函数。

**第一步：在 utils.js 中添加排序函数**

```js
// src/utils.js
export const SORT_BY = {
  CREATED: 'created',
  PRIORITY: 'priority'
};

export const PRIORITY_ORDER = { high: 3, medium: 2, low: 1 };

export function compareTodos(a, b, sortBy) {
  if (sortBy === SORT_BY.PRIORITY) {
    return PRIORITY_ORDER[b.priority] - PRIORITY_ORDER[a.priority];
  }
  return a.createdAt - b.createdAt;
}
```

**第二步：修改 TodoList.jsx**

```jsx
// src/TodoList.jsx
import { SORT_BY, compareTodos } from './utils';

function TodoList({ todos, sortBy = SORT_BY.CREATED }) {
  const sortedTodos = [...todos].sort((a, b) => compareTodos(a, b, sortBy));

  return (
    <ul>
      {sortedTodos.map(todo => (
        <TodoItem key={todo.id} todo={todo} />
      ))}
    </ul>
  );
}
```

**第三步：在 App.jsx 中添加排序控件**

```jsx
// src/App.jsx
import { useState } from 'react';
import TodoList from './TodoList';
import { SORT_BY } from './utils';

function App() {
  const [sortBy, setSortBy] = useState(SORT_BY.CREATED);

  return (
    <div>
      <select value={sortBy} onChange={e => setSortBy(e.target.value)}>
        <option value={SORT_BY.CREATED}>按创建时间</option>
        <option value={SORT_BY.PRIORITY}>按优先级</option>
      </select>
      <TodoList todos={todos} sortBy={sortBy} />
    </div>
  );
}
```
