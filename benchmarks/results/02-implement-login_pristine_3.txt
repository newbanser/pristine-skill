```js
// server.js
const express = require('express');
const sqlite3 = require('sqlite3').verbose();
const crypto = require('crypto');

const app = express();
const db = new sqlite3.Database('./users.db');

app.use(express.json());

// 创建表
db.run(`CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  email TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  salt TEXT NOT NULL
)`);

// 注册接口
app.post('/register', (req, res) => {
  const { email, password } = req.body;
  
  // 生成盐和哈希
  const salt = crypto.randomBytes(16).toString('hex');
  const hash = crypto.pbkdf2Sync(password, salt, 10000, 64, 'sha512').toString('hex');
  
  db.run(
    'INSERT INTO users (email, password_hash, salt) VALUES (?, ?, ?)',
    [email, hash, salt],
    function(err) {
      if (err) {
        if (err.code === 'SQLITE_CONSTRAINT') {
          return res.status(409).json({ error: '邮箱已注册' });
        }
        return res.status(500).json({ error: '服务器错误' });
      }
      res.status(201).json({ id: this.lastID, email });
    }
  );
});

// 登录接口
app.post('/login', (req, res) => {
  const { email, password } = req.body;
  
  db.get('SELECT * FROM users WHERE email = ?', [email], (err, user) => {
    if (err || !user) {
      return res.status(401).json({ error: '邮箱或密码错误' });
    }
    
    const hash = crypto.pbkdf2Sync(password, user.salt, 10000, 64, 'sha512').toString('hex');
    
    if (hash === user.password_hash) {
      const token = crypto.randomBytes(32).toString('hex');
      res.json({ token, email: user.email });
    } else {
      res.status(401).json({ error: '邮箱或密码错误' });
    }
  });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
```

## 依赖

```bash
npm install express sqlite3
```

## 使用

1. 保存为 `server.js`
2. 运行 `node server.js`
3. 注册：`POST /register`，body：`{"email":"test@test.com","password":"123456"}`
4. 登录：`POST /login`，body 同上
