// 基础格式化函数，接受分隔符作为参数
function formatDate(date, separator = '-') {
  const d = new Date(date);
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${d.getFullYear()}${separator}${month}${separator}${day}`;
}

如果希望更加精简，还可以进一步优化：

```javascript
// 使用 options 对象支持更灵活的配置
function formatDate(date, options = {}) {
  const {
    separator = '-',
    format = 'full', // 'full' | 'chinese' | 'short'
    year = true
  } = options;
  
  const d = new Date(date);
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  
  switch (format) {
    case 'chinese':
      return `${d.getMonth() + 1}月${d.getDate()}日`;
    case 'short':
      return year ? `${d.getFullYear()}/${month}/${day}` : `${month}/${day}`;
    default:
      return `${d.getFullYear()}${separator}${month}${separator}${day}`;
  }
}

// 使用示例
formatDate('2024-01-15');                    // "2024-01-15"
formatDate('2024-01-15', { separator: '/' }); // "2024/01/15"
formatDate('2024-01-15', { format: 'chinese' }); // "1月15日"
```

如果你只需要最简化的版本，推荐第一个重构方案，它清晰易懂且保持了原有功能的兼容性。
